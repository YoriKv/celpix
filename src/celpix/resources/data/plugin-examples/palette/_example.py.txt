"""Example palette code format: a 16-step grey ramp, one byte per entry.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one.

A code format implements one palette interpretation directly and appears in the
palette format picker like a preset. Use it for encodings no engine's parameters
express; try a preset first (_color-mask.toml, _color-indexed.toml).
_nes-custom.py is a worked example that loads its colour table from a file.

Contract
- Required methods: ``decode``, ``encode``, ``bytes_per_entry``.
- ``decode`` returns a ``Palette`` of 0xAARRGGBB integers. ``encode`` is its
  inverse; a colour the format cannot store is quantised, not rejected.
- A format with no alpha field must decode alpha as 0xFF. The colour editor
  offers an alpha control when a format decodes alpha other than 0xFF, and
  values set through it would be lost on save.
- Optional: ``entries_per_unit()`` (default 1), the number of entries packed
  into one ``bytes_per_entry`` unit. A handheld shade register holding four
  2-bit shades in a byte returns 4.
- Optional: ``shared_entries(index, count)``, the entries of a ``count``-entry
  palette that show entry ``index``'s colour, owner first. An edit to any of
  them is written to the owner. Default: ``(index,)``.
"""

from celpix.core.palette import Palette
from celpix.plugins import FormatInfo


class Gray4:
    """The top nibble of each byte is a grey level; the low nibble is unused."""

    # `id` is stable and stored in project files; `name` is shown in the picker.
    info = FormatInfo(id="format.palette.example-gray4", name="Example gray (4-bit)")

    def decode(self, data, ctx):
        colors = []
        for b in data:
            v = (b >> 4) * 17  # 0..15 -> 0..255
            colors.append(0xFF000000 | (v << 16) | (v << 8) | v)
        return Palette(colors)

    def encode(self, palette, ctx):
        # The red channel's top nibble: exact for the greys decode produces.
        return bytes(((c >> 16) & 0xF0) for c in palette.colors)

    def bytes_per_entry(self):
        return 1  # sizes a palette read: N entries = N * this many bytes


def register(registry):
    registry.register_format(Gray4())
