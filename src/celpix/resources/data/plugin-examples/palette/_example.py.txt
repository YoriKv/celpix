"""Example palette code format. Copy this file, drop the leading underscore,
edit it, and press F5 in celPix (code plugins ask for confirmation the first
time, since they run with the app's privileges).

A code format implements a palette interpretation directly, for encodings no
built-in engine's parameters can express, and appears in the palette format picker
like any preset. Try a preset first: see _color-mask.toml. decode returns a
Palette of 0xAARRGGBB ints and encode is its exact inverse.

_nes-custom.py is a worked example: a real format that loads its colour table
from a companion file.
"""

from celpix.core.palette import Palette
from celpix.plugins import FormatInfo


class Gray4:
    """One byte per entry, but only the top 4 bits matter: a 16-step gray ramp."""

    # The id doubles as the picker entry's stable identity; the name is shown.
    info = FormatInfo(id="format.palette.example-gray4", name="Example gray (4-bit)")

    def decode(self, data, ctx):
        colors = []
        for b in data:
            v = (b >> 4) * 17  # 0..15 -> 0..255
            # Opaque alpha, and not as a formality: celPix decides whether to
            # offer an alpha input by round-tripping a transparent colour
            # through this format and seeing whether it survives. A format with
            # no alpha field must therefore decode one *back* as 0xFF — leave
            # the byte at 0 and the colour editor offers a slider whose value
            # this format has nowhere to store, so every edit through it is lost
            # on save. Only set alpha from real bits.
            colors.append(0xFF000000 | (v << 16) | (v << 8) | v)
        return Palette(colors)

    def encode(self, palette, ctx):
        # Store the red channel's top nibble; decode's inverse for gray values.
        return bytes(((c >> 16) & 0xF0) for c in palette.colors)

    def bytes_per_entry(self):
        # How many bytes one **read unit** occupies — celPix uses this to size
        # palette reads (e.g. "load 256 entries from here").
        return 1

    # entries_per_unit() is optional and assumed to be 1: define it for a format
    # that packs several entries into one unit, the way a handheld's shade
    # register holds four two-bit shades in a byte. Without it such a format is
    # read one entry per unit, so a window sized for 16 entries covers four.
    # def entries_per_unit(self):
    #     return 4


def register(registry):
    registry.register_format(Gray4())
