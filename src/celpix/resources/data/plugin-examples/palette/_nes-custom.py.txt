"""Worked palette example: NES/Famicom with a user-supplied master palette.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one. ``_example.py`` is the
minimal palette format; read it first.

An NES palette byte is a 6-bit index into the PPU's 64-colour master palette,
whose RGB values differ between emulators. The shipped "NES/Famicom (64-color
indexed)" preset fixes one table; this format loads yours.

Companion file: a 192-byte ``.pal`` (64 entries of R, G, B) beside the plugin,
with the plugin's name: ``nes-custom.py`` reads ``nes-custom.pal``. Without one
the table below is used. The file is read when the plugin loads; press F5 after
changing it.

Decode is a table lookup. Encode selects the nearest entry by RGB distance; the
master palette contains duplicate colours, so an index naming a duplicate is
rewritten as the first match.
"""

from pathlib import Path

from celpix.core.palette import MISSING_COLOR, Palette
from celpix.plugins import FormatInfo

# Fallback master palette (0xRRGGBB), used when no companion .pal is found.
_DEFAULT_MASTER = [
    0x5D5D5D, 0x00218E, 0x120BAF, 0x3900A6, 0x5B0078, 0x6C0033, 0x670500, 0x4E1C00,
    0x293100, 0x044400, 0x004D00, 0x004809, 0x00384F, 0x000000, 0x000000, 0x000000,
    0xA8A8A8, 0x0F54ED, 0x3E33FF, 0x761BFF, 0xA510CE, 0xBD156D, 0xB7290D, 0x934800,
    0x5E6A00, 0x288500, 0x019200, 0x008C30, 0x007595, 0x000000, 0x000000, 0x000000,
    0xFFFFFF, 0x59A7FF, 0x8F83FF, 0xCB67FF, 0xFC5BFF, 0xFF61C1, 0xFF7757, 0xEA990A,
    0xB1BE00, 0x76DA00, 0x48E823, 0x32E17F, 0x38C9EC, 0x454545, 0x000000, 0x000000,
    0xFFFFFF, 0xB9DAFF, 0xD0CBFF, 0xEABFFF, 0xFEBAFF, 0xFFBCE5, 0xFFC6B8, 0xF6D594,
    0xDFE482, 0xC6F087, 0xB2F6A1, 0xA8F3CA, 0xAAE9F7, 0xB0B0B0, 0x000000, 0x000000,
]  # fmt: skip

# A 512-entry (emphasis) .pal is read as its first 64 entries.
_MASTER_COLORS = 64
_PAL_BYTES = _MASTER_COLORS * 3


def _load_master_palette():
    """The 64 master colours: the companion ``.pal`` if present and long enough,
    else the default. ``__file__`` is the plugin's own path."""
    try:
        raw = Path(__file__).with_suffix(".pal").read_bytes()
    except OSError:
        return list(_DEFAULT_MASTER)
    if len(raw) < _PAL_BYTES:  # truncated or not a .pal
        return list(_DEFAULT_MASTER)
    return [
        (raw[i * 3] << 16) | (raw[i * 3 + 1] << 8) | raw[i * 3 + 2]
        for i in range(_MASTER_COLORS)
    ]


class NesCustomPalette:
    """One index byte per entry into a 64-colour table."""

    info = FormatInfo(id="format.palette.nes-custom", name="NES/Famicom (Custom)")

    def __init__(self):
        # Opaque ARGB, built once per load.
        self._table = [0xFF000000 | c for c in _load_master_palette()]

    def decode(self, data, ctx):
        n = len(self._table)
        return Palette([self._table[b] if b < n else MISSING_COLOR for b in data])

    def encode(self, palette, ctx):
        return bytes(self._nearest(argb) for argb in palette.colors)

    def bytes_per_entry(self):
        return 1

    def _nearest(self, argb):
        r, g, b = (argb >> 16) & 0xFF, (argb >> 8) & 0xFF, argb & 0xFF
        best_i, best_d = 0, None
        for i, c in enumerate(self._table):
            d = (
                abs(r - ((c >> 16) & 0xFF))
                + abs(g - ((c >> 8) & 0xFF))
                + abs(b - (c & 0xFF))
            )
            if best_d is None or d < best_d:
                best_i, best_d = i, d
        return best_i


def register(registry):
    registry.register_format(NesCustomPalette())
