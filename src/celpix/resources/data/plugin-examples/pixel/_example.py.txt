"""Example pixel code format: 8x8 tiles at 2bpp, packed.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one.

A code format implements one pixel interpretation directly and appears in the
format picker like a preset. Use it for layouts no engine's parameters express;
this layout is expressible (see _packed.toml) and is used here for brevity.

Contract
- Required methods: ``decode``, ``encode``, ``bytes_per_tile``, ``tile_size``.
- ``decode`` and ``encode`` are stateless and buffer-relative: they process
  whatever bytes they are given, a whole number of tiles, with no knowledge of
  the buffer's position in the file. celPix decodes only the visible window.
- ``encode`` is the exact inverse of ``decode``. Test
  ``encode(decode(x)) == x``; a mismatch corrupts saves.
- An ``IndexGrid`` holds one byte per pixel (a palette index): ``width * height``
  bytes, whatever the format stores on disk. Passing it the raw tile bytes
  raises at decode time unless the format is 8bpp.
- celPix derives the bit depth as
  ``ceil(bytes_per_tile * 8 / (width * height))``. It sizes a palette row and
  clamps painted indices, and it is not checked against ``decode``.
"""

from celpix.core.index_grid import IndexGrid
from celpix.plugins import FormatInfo

# Bit position of each of a byte's four pixels, leftmost first. Shared by decode
# and encode so the two cannot disagree on pixel order.
_SHIFTS = (6, 4, 2, 0)


class Packed2bpp:
    """One byte holds four pixels, leftmost in the high bits. 16 bytes a tile."""

    # `id` is stable and stored in project files; `name` is shown in the picker.
    info = FormatInfo(id="format.pixel.example-2bpp", name="Example packed 2bpp")

    def decode(self, data, ctx):
        """bytes -> list[IndexGrid]. A trailing partial tile is dropped."""
        width, height = self.tile_size()
        size = self.bytes_per_tile()
        tiles = []
        for at in range(0, len(data) - size + 1, size):
            indices = bytearray()  # one per pixel, row-major
            for byte in data[at : at + size]:
                indices += bytes((byte >> shift) & 0x03 for shift in _SHIFTS)
            tiles.append(IndexGrid(width, height, indices))
        return tiles

    def encode(self, tiles, ctx):
        """list[IndexGrid] -> bytes. Indices are masked to the depth: raising
        would make a document unsaveable over one out-of-range pixel."""
        out = bytearray()
        for tile in tiles:
            indices = tile.data
            for at in range(0, len(indices), len(_SHIFTS)):
                byte = 0
                for i, shift in enumerate(_SHIFTS):
                    byte |= (indices[at + i] & 0x03) << shift
                out.append(byte)
        return bytes(out)

    def bytes_per_tile(self):
        return 16  # 8x8 pixels at 2 bits each

    def tile_size(self):
        return (8, 8)  # (width, height) in pixels


def register(registry):
    registry.register_format(Packed2bpp())
