"""Example compression plugin: byte run-length encoding.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one.

Contract
- ``decompress`` runs on load, after the container and reshape stages and before
  the interpret stage. ``compress`` is its exact inverse and runs on save.
- ``compress`` is optional. Without it the data opens read-only.
- ``decompress`` must raise when no complete structure starts at ``data[0]``.
  Scan decodes at each candidate offset and counts a hit when the decode
  succeeds, returns data, and reports ``KEY_DECOMPRESS_COMPLETE``; a lenient
  decoder produces false hits.
- ``KEY_DECOMPRESS_PARTIAL`` is set on the context only by the live preview,
  whose window may cut a structure short. When set, return the valid prefix
  instead of raising. Corrupt data still raises.
- Every ``decompress`` must set two context keys:
  ``KEY_COMPRESSED_SIZE``      bytes consumed; a save-back must fit this size.
  ``KEY_DECOMPRESS_COMPLETE``  whether the end of the structure was reached.

``celpix.plugins.PartialDecompression`` implements both methods for a decoder of
the form ``_decode(data, *, partial) -> (output, consumed, complete)``.

Format: ``(count, value)`` pairs, ``count`` 1-255, ended by a ``count`` of 0.
"""

from celpix.core.context import (
    KEY_COMPRESSED_SIZE,
    KEY_DECOMPRESS_COMPLETE,
    KEY_DECOMPRESS_PARTIAL,
    PipelineContext,
)
from celpix.core.errors import Stage
from celpix.plugins import PluginInfo


class ExampleRle:
    info = PluginInfo(
        id="compression.example-rle",  # stable; stored in project files
        name="Example byte RLE",
        # Optional in a plugin folder, which sets the stage itself.
        stage=Stage.COMPRESSION,
        # True: the stream marks its own end. False: the extent comes from
        # outside (a slice length, a container's byte count). Scan only searches
        # for self-delimiting schemes.
        self_delimiting=True,
        # Also available: `alignment=N`, when the hardware only loads a
        # structure from an N-byte-aligned address. Scan then steps by N.
    )

    def decompress(self, data: bytes, ctx: PipelineContext) -> bytes:
        out = bytearray()
        i, n = 0, len(data)
        complete = False
        while i < n:
            count = data[i]
            if count == 0:  # terminator
                i += 1
                complete = True
                break
            if i + 1 >= n:  # the buffer ends before the value byte
                break
            out += bytes([data[i + 1]]) * count
            i += 2
        if not complete and not ctx.get(KEY_DECOMPRESS_PARTIAL):
            raise ValueError("no terminator: not an Example RLE structure")
        ctx.set(KEY_COMPRESSED_SIZE, i)
        ctx.set(KEY_DECOMPRESS_COMPLETE, complete)
        return bytes(out)

    def compress(self, data: bytes, ctx: PipelineContext) -> bytes:
        out = bytearray()
        i, n = 0, len(data)
        while i < n:
            value = data[i]
            run = 1
            while i + run < n and data[i + run] == value and run < 255:
                run += 1
            out += bytes((run, value))
            i += run
        out.append(0)  # terminator
        return bytes(out)


def register(registry):
    registry.register(ExampleRle())
