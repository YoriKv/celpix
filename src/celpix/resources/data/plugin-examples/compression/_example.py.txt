"""Example compression scheme. Copy this file, drop the leading underscore,
edit it, and press F5 in celPix (code plugins ask for confirmation the first
time, since they run with the app's privileges).

A compression scheme is **one plugin with both directions on it**: ``decompress``
runs *before* the pixel codec, turning a packed byte run into the raw bytes that
format interprets, and ``compress`` is its inverse, run on save so edits write
back packed.

**``compress`` is optional.** A format reverse-engineered far enough to view but
not to re-encode ships ``decompress`` alone, and celPix opens its data read-only
rather than saving something it cannot guarantee. Shipping the method is the
declaration.

This toy scheme is byte run-length encoding: a series of ``(count, value)`` pairs,
``count`` in 1..255, terminated by a ``count`` of 0. Real schemes are more
involved, but the plumbing is identical.
"""

from celpix.core.context import (
    KEY_COMPRESSED_SIZE,
    KEY_DECOMPRESS_COMPLETE,
    KEY_DECOMPRESS_PARTIAL,
    PipelineContext,
)
from celpix.core.errors import Stage
from celpix.plugins import PluginInfo


class ExampleRle:
    info = PluginInfo(
        id="compression.example-rle",
        name="Example byte RLE",
        stage=Stage.COMPRESSION,
        # A property of the *scheme*, about the decode direction: this one carries
        # a terminator, so its extent is knowable from the data alone. Set False
        # for a scheme that ends only where something outside it says — a slice
        # length, a container's byte count. celPix words "the decode stopped here"
        # differently for the two, and Scan can only find the first kind.
        self_delimiting=True,
    )

    def decompress(self, data: bytes, ctx: PipelineContext) -> bytes:
        """Expand ``(count, value)`` pairs until the 0-count terminator.

        **Strict by default: raise when the structure does not end inside the
        buffer.** That is what makes "these bytes are an Example RLE structure" a
        real question — Scan tries a decode at every offset and treats success as
        a hit, so a best-effort decompressor reports a structure everywhere.

        ``KEY_DECOMPRESS_PARTIAL`` is set when, and only when, celPix knowingly
        hands over a window that may cut a structure short (the live preview).
        Honour it by returning the valid prefix; corrupt data still raises.

        Two things go back through ``ctx``:

        - ``KEY_COMPRESSED_SIZE`` — bytes consumed, the packed structure's true
          length, which a save-back must fit.
        - ``KEY_DECOMPRESS_COMPLETE`` — whether the terminator was reached inside
          the buffer (False = the window cut the stream short).
        """
        out = bytearray()
        i, n = 0, len(data)
        complete = False
        while i < n:
            count = data[i]
            if count == 0:  # terminator
                i += 1
                complete = True
                break
            if i + 1 >= n:  # window ended before the value byte
                break
            out += bytes([data[i + 1]]) * count
            i += 2
        if not complete and not ctx.get(KEY_DECOMPRESS_PARTIAL):
            raise ValueError("no terminator — no Example RLE structure here")
        ctx.set(KEY_COMPRESSED_SIZE, i)
        ctx.set(KEY_DECOMPRESS_COMPLETE, complete)
        return bytes(out)

    def compress(self, data: bytes, ctx: PipelineContext) -> bytes:
        """Pack ``data`` into ``(count, value)`` pairs — the inverse of decode.

        Runs longer than 255 split across successive pairs, and the stream ends
        with a 0-count terminator so decode knows where it stops.
        """
        out = bytearray()
        i, n = 0, len(data)
        while i < n:
            value = data[i]
            run = 1
            while i + run < n and data[i + run] == value and run < 255:
                run += 1
            out.append(run)
            out.append(value)
            i += run
        out.append(0)  # terminator
        return bytes(out)


def register(registry):
    registry.register(ExampleRle())
