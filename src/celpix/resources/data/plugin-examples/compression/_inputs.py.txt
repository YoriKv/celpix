"""Worked compression example: a codec that reads data from outside its stream.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one. ``_example.py`` is the
minimal compression plugin; read it first.

A stage is handed one byte range. A format that also depends on data elsewhere
(a table shared by many streams, a count the file does not store) declares an
``InputSpec`` for each item. The user binds each input per entry in the Inputs
window: File > Inputs..., or the badge beside the Compression and Tilemap
pickers. celPix resolves the bindings and delivers the values, read-only, in
``ctx.get(KEY_INPUTS)`` as a dict keyed by ``InputSpec.key``.

Rules
- Only compression plugins and tilemap code formats may declare inputs. A
  tilemap format passes them as ``FormatInfo(..., inputs=(...))``. Declared on
  any other stage they are ignored, and celPix warns at load.
- Kinds: ``REGION`` (bytes), ``INTEGER`` and ``FLAG`` (bool).
- A required input is always present when a method runs. If it is unbound or
  invalid, celPix runs the stage as pass-through, marks the entry and names the
  input to bind.
- An optional input that is unbound and has no ``default`` is absent from the
  dict. A flag is always present: ``bool(default)`` when unbound.
- ``KEY_INPUTS`` is set for ``decompress``, ``compress``, the live preview and
  Scan.
- Inputs are never written back. ``compress`` must raise for data the bound
  inputs cannot represent.
- ``InputSpec.key`` is stored in project files: renaming it orphans bindings.

Format: byte-pair encoding. A byte below ``first_code`` is a literal; a byte at
or above it selects a two-byte phrase from a table stored elsewhere in the file.
"""

from celpix.core.context import (
    KEY_COMPRESSED_SIZE,
    KEY_DECOMPRESS_COMPLETE,
    KEY_INPUTS,
    PipelineContext,
)
from celpix.core.errors import Stage
from celpix.plugins import InputKind, InputSpec, PluginInfo

_PHRASE = 2  # bytes one code expands to
_DEFAULT_FIRST_CODE = 0x80


class ExampleBytePair:
    info = PluginInfo(
        id="compression.example-byte-pair",
        name="Example byte-pair (shared phrase table)",
        stage=Stage.COMPRESSION,
        # No end marker: the slice sets the extent, and Scan does not apply.
        self_delimiting=False,
        inputs=(
            InputSpec(
                "table",  # key
                "Phrase table",  # label shown in the Inputs window
                InputKind.REGION,
                # Element size in bytes. A bound length must be a multiple of
                # it, and the UI counts the region in `unit`s.
                stride=_PHRASE,
                unit="phrase",
                # Qt does not wrap tooltips: break lines at about 60 characters.
                tooltip=(
                    "The two-byte phrases the codes name, in code\n"
                    "order. Shared by every stream in the file, so\n"
                    "bind it to the ROM rather than copying it."
                ),
            ),
            InputSpec(
                "first_code",
                "First code",
                InputKind.INTEGER,
                required=False,
                # Delivered when the input is unbound, so the common case needs
                # no binding.
                default=_DEFAULT_FIRST_CODE,
                # A bound value outside this range is rejected.
                minimum=1,
                maximum=0xFF,
                tooltip=(
                    "The packed byte value at which phrase codes start.\n"
                    "Below it a byte is a literal. 0x80 unless the game\n"
                    "kept a bigger literal alphabet."
                ),
            ),
        ),
    )

    def decompress(self, data: bytes, ctx: PipelineContext) -> bytes:
        inputs = ctx.get(KEY_INPUTS) or {}
        table: bytes = inputs["table"]  # required: always present
        first: int = inputs["first_code"]  # has a default: always present
        out = bytearray()
        for byte in data:
            if byte < first:
                out.append(byte)
                continue
            at = (byte - first) * _PHRASE
            if at + _PHRASE > len(table):
                raise ValueError(f"code 0x{byte:02X} is past the phrase table")
            out += table[at : at + _PHRASE]
        ctx.set(KEY_COMPRESSED_SIZE, len(data))
        # No end marker, so the structure is never known to be complete.
        ctx.set(KEY_DECOMPRESS_COMPLETE, False)
        return bytes(out)

    def compress(self, data: bytes, ctx: PipelineContext) -> bytes:
        """Greedy, so the output may differ from the original packed bytes. The
        requirement is ``decompress(compress(x)) == x``."""
        inputs = ctx.get(KEY_INPUTS) or {}
        table: bytes = inputs["table"]
        first: int = inputs["first_code"]
        codes = {
            table[at : at + _PHRASE]: first + at // _PHRASE
            for at in range(0, len(table) - _PHRASE + 1, _PHRASE)
            if first + at // _PHRASE <= 0xFF
        }
        out = bytearray()
        i = 0
        while i < len(data):
            code = codes.get(data[i : i + _PHRASE])
            if code is not None:
                out.append(code)
                i += _PHRASE
            elif data[i] < first:
                out.append(data[i])
                i += 1
            else:
                # The table is read-only, so this data cannot be stored.
                raise ValueError(
                    f"0x{data[i]:02X} at {i} is neither a literal nor a phrase "
                    "in the table - the edit cannot be written back"
                )
        return bytes(out)


def register(registry):
    registry.register(ExampleBytePair())
