"""Example reshape plugin: swap the two halves of a region.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one.

A reshape is a length-preserving reordering of a whole region, such as joining
an arcade board's per-chip bitplanes or de-interleaving a ROM pair. It runs
after the container and before decompression.

Contract
- ``reshape`` runs on load and receives the entire region, never a window.
- ``unshape`` is its exact inverse and runs on save. It is optional; without it
  the data opens read-only.
- The output length must equal the input length.
- While a reshape is active, view addresses count from the start of the
  reordered buffer, not the file.
- Test ``unshape(reshape(x)) == x`` for odd and even lengths. The pair is not
  checked at runtime, and a mismatch corrupts saves.

An address-line permutation or a byte-value substitution needs no code: see
_bitswap.toml and _data-lut.toml.
"""

from celpix.core.context import PipelineContext
from celpix.core.errors import Stage
from celpix.plugins import PluginInfo


class ExampleSwapHalves:
    info = PluginInfo(
        id="reshape.example-swap-halves",  # stable; stored in project files
        name="Example half swap",
        # Optional in a plugin folder, which sets the stage itself.
        stage=Stage.RESHAPE,
    )

    def reshape(self, data: bytes, ctx: PipelineContext) -> bytes:
        # An odd trailing byte has no partner and passes through unchanged.
        half = len(data) // 2
        return data[half : half * 2] + data[:half] + data[half * 2 :]

    def unshape(self, data: bytes, ctx: PipelineContext) -> bytes:
        return self.reshape(data, ctx)  # a half swap is its own inverse


def register(registry):
    registry.register(ExampleSwapHalves())
