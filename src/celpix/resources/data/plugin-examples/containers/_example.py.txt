"""Example container plugin: an 8-byte magic header in front of the payload.

To use: copy, remove the leading "_", edit, press F5. Code plugins run with the
app's privileges, so celPix asks before loading one. ``_tiff.py`` is a worked
example on a real format.

A container removes on-disk framing (a header, an interleave, a checksum block)
on load and restores it on save.

Contract
- ``read(source, ctx)`` returns the payload. ``write(data, dest, ctx)`` returns
  the destination file's complete new contents. Neither touches the disk: the
  host supplies the bytes (``ReadSource``, ``WriteTarget``) and writes the result.
- ``write`` is optional. Without it the container's files open read-only.
- ``read`` must set ``KEY_SOURCE_OFFSET``: the file-absolute offset at which the
  returned payload starts. ``source.base`` is the file offset of
  ``source.data[0]``.
- ``source.offset`` / ``source.length``: the caller wants a bounded window.
  A container that strips nothing can return ``plain_read(source, ctx)``.
- ``dest.offset`` / ``dest.length``: the target is a slot inside a larger file.
  Returning only the payload would truncate the file; use ``splice``.
- ``dest.existing`` is empty for Save As and for File > New File.... The
  framing must then be rebuilt, or (Save As only) restored from data ``read``
  stored on the context, as ``_tiff.py`` does.
- ``describe(source, ctx)`` is optional: rows for the Container Info popup.
"""

from celpix.core.capabilities import ContentKind
from celpix.core.context import KEY_SOURCE_OFFSET, PipelineContext
from celpix.core.errors import Stage
from celpix.plugins import (
    ContainerField,
    PluginInfo,
    ReadSource,
    WriteTarget,
    splice,
)

_MAGIC = b"CELPIXEX"


class ExampleContainer:
    info = PluginInfo(
        id="container.example",  # stable; stored in project files
        name="Example container (magic header)",
        # Optional in a plugin folder, which sets the stage itself.
        stage=Stage.CONTAINER,
        # Detection is declarative and runs without executing plugin code.
        #   magic       (offset, bytes) probes; any one matching claims the file.
        #   extensions  lowercase suffixes, e.g. (".bin",). Used for detection
        #               only when no magic is declared; also the default suffix
        #               in File > New File....
        #   exact_size, size_modulo, min_size
        #               narrow a magic or extension match; never match alone.
        magic=((0, _MAGIC),),
        # Shown beside the filename in the Files list: "art.bin (Example)".
        short_name="Example",
        # The entry kinds this container frames. This pair is the default. A
        # palette file format uses (ContentKind.PALETTE,).
        content_kinds=(ContentKind.PIXELS, ContentKind.TILEMAP),
        # True: every payload byte keeps its file position (the container only
        # skips bytes), so slices can address the file directly. celPix checks
        # the claim on each read and warns on a mismatch. Default False: required
        # for a container that reorders or joins bytes.
        preserves_offsets=True,
    )

    def read(self, source: ReadSource, ctx: PipelineContext) -> bytes:
        # The payload start is the format's own, so source.offset is not used.
        raw = source.data
        start = len(_MAGIC) if raw[: len(_MAGIC)] == _MAGIC else 0
        ctx.set(KEY_SOURCE_OFFSET, source.base + start)
        end = len(raw) if source.length is None else start + source.length
        return raw[start:end]

    def write(self, data: bytes, dest: WriteTarget, ctx: PipelineContext) -> bytes:
        # Whole file, including an empty destination: rebuild the framing. This
        # header is constant; one that records a length, offsets or a checksum
        # must be regenerated here.
        if dest.whole_file:
            return _MAGIC + data
        # A slot inside an existing file: keep everything around it.
        return splice(dest.existing, dest.offset, data)

    def describe(
        self, source: ReadSource, ctx: PipelineContext
    ) -> tuple[ContainerField, ...]:
        # Display only. An exception here discards the rows and nothing else.
        # `detail` is a tooltip: Qt does not wrap it, so break lines at about 60
        # characters.
        present = source.data[: len(_MAGIC)] == _MAGIC
        return (
            ContainerField(
                "Magic header",
                f"{_MAGIC.decode()} at 0x000000" if present else "absent",
                "The eight bytes this container frames its payload with.\n"
                "Absent, the whole file is treated as payload.",
            ),
        )


def register(registry):
    registry.register(ExampleContainer())
