"""Example container format. Copy this file, drop the leading underscore, edit
it, and press F5 in celPix (code plugins ask for confirmation the first time,
since they run with the app's privileges).

A container is **one plugin with both directions on it**: ``read`` strips the
framing the pixel codec shouldn't see — a header, an interleave, a checksum block
— and ``write`` re-wraps the edited bytes on save. **``write`` is optional**, and
a container without one is view-only: its files open and cannot be saved.

This toy container is a fixed 8-byte magic header (``CELPIXEX``) in front of the
payload: ``read`` skips it, ``write`` puts it back. See _tiff.py for a worked
example on a real format.

**Neither direction touches the disk.** The host hands ``read`` the bytes
(``ReadSource``), and ``write`` is shown the destination as it stands
(``WriteTarget``) and *returns* what should replace it — so a container works
whether its bytes came from a file, a parent's unsaved buffer, or several files
joined together.

Three cases separate a container that works from one that quietly eats data:

- ``source.offset`` / ``source.length`` — the caller wants a bounded window, not
  everything to EOF. ``source.window()`` applies both, for a container with no
  opinion about where the payload starts.
- ``dest.offset`` / ``dest.length`` — the save target is a slot *inside* a larger
  file. Returning just the payload truncates the file to it; ``splice`` lays the
  edit over ``dest.existing`` and keeps everything around it.
- ``dest.existing`` is **empty** — a Save As, to a path with nothing at it yet.
  There is no file to splice into, so the framing must be rebuilt (this one can:
  the header is a constant) or carried over from the read on the context. Write
  the payload alone and you produce a file that will not reopen as your format.
  ``_tiff.py`` shows the stash; celPix's own iNES plugin does the same.

``describe`` is a fourth, optional method: rows for the container-info popup,
display-only and reached with ``getattr``.
"""

from celpix.core.capabilities import ContentKind
from celpix.core.context import KEY_SOURCE_OFFSET, PipelineContext
from celpix.core.errors import Stage
from celpix.plugins import (
    ContainerField,
    PluginInfo,
    ReadSource,
    WriteTarget,
    splice,
)

_MAGIC = b"CELPIXEX"  # 8-byte header this toy container wraps its payload in


class ExampleContainer:
    info = PluginInfo(
        id="container.example",
        name="Example container (magic header)",
        stage=Stage.CONTAINER,
        # Detection is declarative: describe the file, never inspect it, and the
        # container claims its own files instead of being picked by hand.
        # `extensions` is the other half — see _tiff.py, which claims .tif/.tiff
        # by name as well as by both byte orders. `size_modulo` and `min_size`
        # never make a match, only narrow one.
        magic=((0, _MAGIC),),
        # The tag the Files list shows beside a filename: "art.bin (Example)".
        short_name="Example",
        # What kind of entry this frames — detection and the container picker
        # both read it, so a palette is never offered a ROM wrapper or the
        # reverse. The pair below *is* the default (pixels and tilemaps), spelled
        # out because the line most people want is the other one,
        # `content_kinds=(ContentKind.PALETTE,)`.
        content_kinds=(ContentKind.PIXELS, ContentKind.TILEMAP),
    )

    def read(self, source: ReadSource, ctx: PipelineContext) -> bytes:
        """Return the payload after the magic header (or the whole file if absent).

        Publish the byte offset the returned data starts at — the one thing only
        the container knows — so ``write`` can splice edits back into the right
        place and the UI can show real file addresses. It is **file-absolute**,
        hence ``source.base``: the buffer can begin part way into the file, as
        when a slice reads its parent's unsaved bytes.

        Where the payload begins is the *container's* answer, so ``source.offset``
        is not consulted here. A container stripping no framing at all wants
        ``plain_read(source, ctx)`` instead (from ``celpix.plugins``), which
        publishes the offset and windows the bytes in one line.
        """
        raw = source.data
        start = len(_MAGIC) if raw[: len(_MAGIC)] == _MAGIC else 0
        ctx.set(KEY_SOURCE_OFFSET, source.base + start)
        end = len(raw) if source.length is None else start + source.length
        return raw[start:end]

    def write(self, data: bytes, dest: WriteTarget, ctx: PipelineContext) -> bytes:
        """Re-wrap ``data`` in the magic header — the inverse of ``read``.

        Only the whole-file case re-wraps, the framing *being* the file. A bounded
        target is a slot inside a file that already has framing, so the payload is
        spliced into what is there; always returning ``magic + data`` would
        truncate the file to one slice.

        The whole-file branch is also what makes **Save As** right: an empty
        ``dest.existing`` is still a whole-file target, and this container can
        rebuild its framing because the framing is a constant. Check that for your
        own format — a header recording anything about the payload (a length, an
        offset table, a checksum) has to be regenerated here or carried over from
        the read.
        """
        if dest.whole_file:
            return _MAGIC + data
        return splice(dest.existing, dest.offset, data)

    def describe(
        self, source: ReadSource, ctx: PipelineContext
    ) -> tuple[ContainerField, ...]:
        """Optional: what this container found, for the container-info popup.

        Display-only and reached with ``getattr``, so a container without one
        reports nothing rather than breaking. ``detail`` becomes a tooltip — hard-
        wrap it near 60 characters, because Qt will not wrap it for you.
        """
        present = source.data[: len(_MAGIC)] == _MAGIC
        return (
            ContainerField(
                "Magic header",
                f"{_MAGIC.decode()} at 0x000000" if present else "absent",
                "The eight bytes this container frames its payload with.\n"
                "Absent, the whole file is treated as payload - which is\n"
                "what lets a half-written file still open.",
            ),
        )


def register(registry):
    registry.register(ExampleContainer())
