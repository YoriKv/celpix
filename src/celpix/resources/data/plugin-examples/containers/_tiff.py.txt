"""Worked container example: baseline TIFF (``.tif`` / ``.tiff``).

Files starting with "_" are ignored — copy this file, drop the underscore from
the name, and press F5 in celPix to load it (code plugins ask for confirmation
the first time, since they run with the app's privileges).

``_example.py`` alongside this file is the minimal shape. This is a real format,
and a good one to learn containers from because its framing is a *lookup* rather
than a fixed header. A TIFF does not store its pixels at a known offset: it stores
a directory (the IFD) whose entries say where the image data sits, in one or more
"strips" scattered anywhere in the file. So ``read`` parses the directory and
hands celPix the strips joined end to end, and ``write`` puts edited bytes back
strip by strip — the general container job, where the byte range the codec edits
has no fixed relationship to the file.

**Layout.** ``II``/``MM`` picks the byte order for every number in the file, then
magic 42, then the offset of the first IFD. An IFD is an entry count followed by
12-byte entries — tag, type, count, and either the value itself when it fits in
four bytes or an offset to it. Three tags matter here: ``StripOffsets`` (273),
``StripByteCounts`` (279) and ``Compression`` (259).

**Compression.** Baseline TIFF's own PackBits is a *celPix* compression plugin,
not a container concern: this hands over the strips exactly as stored and reports
which scheme they use, and the user picks "PackBits" in the Compression dropdown
to unpack them. The two layers compose, which is why they are separate stages.

**Telling the user something.** Two channels. Raise when the read cannot proceed:
celPix stops, reports it, and nothing opens. Call ``warn``/``inform`` when it
*can* proceed but the result deserves a caveat — an unsupported compression,
strips joined out of a scattered file. Those mark the entry in the files list and
spell themselves out in its tooltip without interrupting anyone. A plugin may also
record notices and *then* raise, so a hard failure carries what led up to it.

**What it doesn't do**, being an example: tiled TIFFs (``TileOffsets``, tag 324),
multi-page files (only the first IFD is read), BigTIFF, and any compression beyond
none/PackBits. Each is more parsing, not a different shape.
"""

from celpix.core.context import KEY_SOURCE_OFFSET, PipelineContext
from celpix.core.errors import Stage
from celpix.core.notices import inform, warn
from celpix.plugins import PluginInfo, ReadSource, WriteTarget

_LITTLE = b"II*\x00"  # byte order + magic 42, together the whole signature
_BIG = b"MM\x00*"

_TAG_COMPRESSION = 259
_TAG_STRIP_OFFSETS = 273
_TAG_STRIP_BYTE_COUNTS = 279

# The IFD value types this container understands. The three tags it wants are all
# counts and offsets, so BYTE/SHORT/LONG cover them; an entry of any other type
# is skipped rather than guessed at.
_TYPE_SIZES = {1: 1, 3: 2, 4: 4}

_COMPRESSION_NAMES = {1: "none", 32773: "PackBits"}


def _entry(raw: bytes, order: str, at: int) -> tuple[int, list[int]]:
    """One 12-byte IFD entry as ``(tag, values)``; unknown types yield no values."""
    tag = int.from_bytes(raw[at : at + 2], order)
    type_id = int.from_bytes(raw[at + 2 : at + 4], order)
    count = int.from_bytes(raw[at + 4 : at + 8], order)
    size = _TYPE_SIZES.get(type_id, 0)
    if size == 0 or count == 0:
        return tag, []
    # The last four bytes are the value itself when it fits, and a pointer to it
    # when it doesn't — the piece of TIFF that catches everybody out.
    field = at + 8
    start = field if size * count <= 4 else int.from_bytes(raw[field : field + 4], order)
    return tag, [
        int.from_bytes(raw[start + i * size : start + (i + 1) * size], order)
        for i in range(count)
    ]


def _strips(raw: bytes) -> tuple[list[tuple[int, int]], int]:
    """``([(offset, length), …], compression)`` for the first image in ``raw``."""
    if raw[:4] == _LITTLE:
        order = "little"
    elif raw[:4] == _BIG:
        order = "big"
    else:
        raise ValueError("not a TIFF: no II*/MM* signature")
    ifd_at = int.from_bytes(raw[4:8], order)
    tags: dict[int, list[int]] = {}
    for i in range(int.from_bytes(raw[ifd_at : ifd_at + 2], order)):
        tag, values = _entry(raw, order, ifd_at + 2 + i * 12)
        tags[tag] = values
    offsets = tags.get(_TAG_STRIP_OFFSETS, [])
    counts = tags.get(_TAG_STRIP_BYTE_COUNTS, [])
    if not offsets or len(offsets) != len(counts):
        raise ValueError("no usable StripOffsets/StripByteCounts in the first IFD")
    compression = (tags.get(_TAG_COMPRESSION) or [1])[0]
    return list(zip(offsets, counts)), compression


class TiffContainer:
    info = PluginInfo(
        id="container.tiff",
        name="TIFF image (strip data)",
        stage=Stage.CONTAINER,
        extensions=(".tif", ".tiff"),
        # Both byte orders, so a TIFF is claimed whichever machine wrote it.
        magic=((0, _LITTLE), (0, _BIG)),
        # The tag the Files list shows beside a filename: "board.tif (TIFF)".
        short_name="TIFF",
        # The strips are joined end to end, so a position in the result names no
        # position in the file. Declaring that stops celPix carving a slice, or
        # writing an offset palette, at coordinates meaning nothing here: the
        # read reorders, so only this container can place a write.
        preserves_offsets=False,
    )

    def read(self, source: ReadSource, ctx: PipelineContext) -> bytes:
        """The image strips, joined — the bytes a pixel codec should see."""
        # The host has already acquired the file's bytes, so a container is a
        # pure transform of them and never opens source.path.
        raw = source.data
        strips, compression = _strips(raw)
        # Provenance is the first strip: where the data celPix now holds begins in
        # the file, which is what the address display and a save-back want. The
        # strips after it are not contiguous with it, which is why the write
        # below cannot be a single splice.
        ctx.set(KEY_SOURCE_OFFSET, strips[0][0])
        name = _COMPRESSION_NAMES.get(compression, f"unsupported ({compression})")
        # Advisory, like everything on the context: it tells the user which
        # Compression entry to pick without this container deciding for them.
        ctx.set("tiff.compression", name)
        # A *notice* is the other half of that. The context entry above is for
        # another stage to read; this is for the person looking at the result.
        # Use it when the read succeeded but the bytes are not quite what was
        # asked for — celPix marks the entry in the files list and puts them in
        # its tooltip. Raise instead when the read cannot proceed (see _strips).
        if compression not in _COMPRESSION_NAMES:
            warn(
                ctx,
                f"Unsupported TIFF compression ({compression})",
                "The strips are handed over exactly as stored, so they\n"
                "will look like noise until a matching Compression\n"
                "plugin unpacks them - and this example ships none.",
                self.info.id,
            )
        if len(strips) > 1:
            inform(
                ctx,
                f"Joined {len(strips)} strips",
                "The image is stored in pieces scattered through the\n"
                "file. They are shown end to end here, and written back\n"
                "to their own places on save.",
                self.info.id,
            )
        joined = b"".join(raw[at : at + length] for at, length in strips)
        if source.length is not None:
            joined = joined[: source.length]
        return joined

    def write(self, data: bytes, dest: WriteTarget, ctx: PipelineContext) -> bytes:
        """Put edited bytes back into the strips they were read out of.

        ``dest.offset`` is not consulted: the strips say where the bytes go and
        they are not contiguous, so a single file offset could not describe the
        destination even in principle. Every container that relocates data ends up
        here, the read's mapping being the only thing that knows how to undo
        itself. Returning the file *whole* is what makes that expressible — the
        edit lands in several places at once, and everything between the strips is
        carried across untouched.

        The strips are a fixed set of slots, so the edited data has to be exactly
        the size it was read as. It always is for an ordinary tile edit, which
        rewrites bytes in place; a re-compression that packed to a different size
        would not fit, and is refused rather than written half-way.
        """
        strips, _compression = _strips(dest.existing)
        total = sum(length for _at, length in strips)
        if len(data) != total:
            raise ValueError(
                f"{len(data)} bytes do not fit the image's {total}-byte strips"
            )
        out = bytearray(dest.existing)
        taken = 0
        for at, length in strips:
            out[at : at + length] = data[taken : taken + length]
            taken += length
        return bytes(out)


def register(registry):
    registry.register(TiffContainer())
